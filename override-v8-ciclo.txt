
    /* ===== MEDCID v8: ciclo de estudos ===== */
    const CICLO_HEAT_RANK={'Muito cobrado':0,'Cobrado':1,'Não classificado':2,'Ocasional':3,'Raro':4};
    let cicloClockId=null, cicloAudio=null, cicloResetArmed=false;

    function cicloState(){
      state.cycle=state.cycle&&typeof state.cycle==='object'?state.cycle:{};
      const c=state.cycle;
      if(!Array.isArray(c.blocks))c.blocks=[];
      if(!Array.isArray(c.log))c.log=[];
      c.pos=Math.max(0,Number(c.pos||0));
      if(!c.timer||typeof c.timer!=='object')c.timer={running:false,endAt:null,remaining:null,total:null,blockId:null};
      if(c.blocks.length)c.pos%=c.blocks.length;else c.pos=0;
      return c;
    }
    function cicloSubject(id){return state.subjects.find(s=>String(s.id)===String(id));}
    function cicloTopics(subject){const all=subject?.topics||[];return subject?.phase==='AP2'?all.filter(t=>t.phase==='AP2'):all;}
    function cicloAP1Grade(subject){
      const arch=state.phaseArchives?.AP1_2026_2?.recovery?.assessments?.[String(subject.id)];
      const g=Number(arch?.grade||0);return g>0?g:null;
    }
    function cicloCurrentBlock(){const c=cicloState();return c.blocks[c.pos]||null;}

    function cicloBuildDefault(){
      let subs=state.subjects.filter(s=>cicloTopics(s).length);
      if(!subs.length)subs=[...state.subjects];
      const pool=subs.map(s=>{
        const pending=cicloTopics(s).filter(t=>!t.done).length;
        const count=Math.max(1,Math.min(3,Math.round(pending/5)||1));
        const g=cicloAP1Grade(s);
        return {s,count,minutes:g!==null&&g<7?90:60};
      });
      const blocks=[];let last=null;
      while(pool.some(p=>p.count>0)){
        const options=pool.filter(p=>p.count>0).sort((a,b)=>b.count-a.count);
        const pick=options.find(p=>p.s.id!==last)||options[0];
        blocks.push({id:uid(),subjectId:String(pick.s.id),minutes:pick.minutes});
        pick.count--;last=pick.s.id;
      }
      return blocks;
    }
    function cicloEnsureBlocks(){
      const c=cicloState();
      if(!c.blocks.length&&state.subjects.length){c.blocks=cicloBuildDefault();c.pos=0;cicloResetTimer(false);return true;}
      c.blocks=c.blocks.filter(b=>cicloSubject(b.subjectId));
      if(c.blocks.length)c.pos%=c.blocks.length;else c.pos=0;
      return false;
    }

    function cicloSuggest(subject){
      if(!subject)return null;
      const topics=cicloTopics(subject),agg=aggregateQuestions();
      const rateOf=t=>{const x=agg[normalize(subject.name+'|'+t.name)];return x&&x.total?x.rate:null;};
      const weak=topics.map(t=>({t,r:rateOf(t)})).filter(x=>x.r!==null&&x.r<70).sort((a,b)=>a.r-b.r)[0];
      if(weak)return {topic:weak.t,kind:'Questões',why:`${weak.r}% de acerto nas questões registradas`};
      const pending=topics.filter(t=>!t.done).sort((a,b)=>(CICLO_HEAT_RANK[a.heat]??2)-(CICLO_HEAT_RANK[b.heat]??2));
      if(pending[0])return {topic:pending[0],kind:'Teoria + questões',why:pending[0].heat==='Muito cobrado'?'assunto muito cobrado ainda não concluído':'próximo assunto pendente'};
      if(topics.length)return {topic:null,kind:'Provas antigas',why:'todos os assuntos da fase concluídos'};
      return {topic:null,kind:'Estudo livre',why:'nenhum assunto cadastrado nesta disciplina'};
    }

    function cicloMinutesWeek(subjectId){
      const start=addDays(todayISO(),-6);
      return cicloState().log.filter(l=>String(l.subjectId)===String(subjectId)&&l.date>=start).reduce((a,l)=>a+Number(l.minutes||0),0);
    }
    function cicloExamDays(subject){
      const exam=subjectAssessment(subject).exam;if(!exam)return null;
      return Math.round((parseISO(exam)-parseISO(todayISO()))/86400000);
    }

    /* ---- cronômetro ---- */
    function cicloRemaining(){
      const t=cicloState().timer;
      if(t.running&&t.endAt)return Math.max(0,Math.ceil((Number(t.endAt)-Date.now())/1000));
      return Math.max(0,Number(t.remaining??t.total??0));
    }
    function cicloResetTimer(persist=true){
      const c=cicloState(),b=c.blocks[c.pos];
      c.timer={running:false,endAt:null,total:b?Number(b.minutes)*60:0,remaining:b?Number(b.minutes)*60:0,blockId:b?.id||null};
      if(persist)save();
    }
    function cicloToggleTimer(){
      const c=cicloState();if(!c.blocks.length)return;
      const t=c.timer;
      if(t.blockId!==c.blocks[c.pos].id)cicloResetTimer(false);
      if(t.running){t.remaining=cicloRemaining();t.running=false;t.endAt=null;}
      else{t.endAt=Date.now()+cicloRemaining()*1000;t.running=true;}
      save();renderCiclo();
    }
    function cicloBeep(){
      try{cicloAudio=cicloAudio||new (window.AudioContext||window.webkitAudioContext)();const o=cicloAudio.createOscillator(),g=cicloAudio.createGain();o.frequency.value=880;g.gain.value=.08;o.connect(g);g.connect(cicloAudio.destination);o.start();o.stop(cicloAudio.currentTime+.7);}catch(e){}
      try{if('Notification' in window&&Notification.permission==='granted')new Notification('MEDCID · Bloco concluído',{body:'Registre o assunto estudado e siga para o próximo bloco.'});}catch(e){}
    }
    function cicloConclude(auto){
      const c=cicloState(),b=c.blocks[c.pos];if(!b)return;
      const t=c.timer,total=Number(t.total||b.minutes*60),left=cicloRemaining();
      const minutes=t.blockId===b.id?Math.round((total-left)/60):0;
      const subject=cicloSubject(b.subjectId);
      const topicId=$('#cicloTopic')?.value||'';
      const topic=cicloTopics(subject).find(x=>x.id===topicId)||null;
      const markDone=!!$('#cicloMarkDone')?.checked;
      if(minutes>0){
        c.log.push({id:uid(),date:todayISO(),subjectId:String(b.subjectId),minutes,topicId:topic?.id||null,topic:topic?.name||null});
        if(c.log.length>1500)c.log=c.log.slice(-1500);
        if(minutes>=25)adjustXP(10);
      }
      c.pos=(c.pos+1)%c.blocks.length;
      cicloResetTimer(false);
      if(markDone&&topic&&!topic.done){toggleTopic(subject.id,topic.id);}else{save();renderAll();}
      if(auto)cicloBeep();
      toast(minutes>0?`${subject?.name||'Bloco'}: ${minutes} min registrados${minutes>=25?' · +10 XP':''}.`:'Bloco encerrado sem tempo registrado.');
    }
    function cicloSkip(){const c=cicloState();if(!c.blocks.length)return;c.pos=(c.pos+1)%c.blocks.length;cicloResetTimer(false);save();renderCiclo();}
    function cicloTick(){
      const c=cicloState(),t=c.timer;
      if(t.running&&cicloRemaining()<=0){cicloConclude(true);return;}
      const s=cicloRemaining(),txt=String(Math.floor(s/60)).padStart(2,'0')+':'+String(s%60).padStart(2,'0');
      const el=$('#cicloTime');if(el)el.textContent=txt;
      const mini=$('#cicloDashTime');if(mini)mini.textContent=t.running?txt:'';
      const bar=$('#cicloTimeBar');if(bar)bar.style.width=(t.total?100*(1-s/t.total):0)+'%';
    }
    function startCicloClock(){if(cicloClockId)clearInterval(cicloClockId);cicloTick();cicloClockId=setInterval(cicloTick,1000);}

    /* ---- edição dos blocos ---- */
    function cicloMove(id,dir){
      const c=cicloState(),i=c.blocks.findIndex(b=>b.id===id),j=i+dir;
      if(i<0||j<0||j>=c.blocks.length)return;
      const current=c.blocks[c.pos]?.id;
      [c.blocks[i],c.blocks[j]]=[c.blocks[j],c.blocks[i]];
      c.pos=Math.max(0,c.blocks.findIndex(b=>b.id===current));
      save();renderCiclo();
    }
    function cicloRemove(id){
      const c=cicloState(),current=c.blocks[c.pos]?.id;
      c.blocks=c.blocks.filter(b=>b.id!==id);
      const k=c.blocks.findIndex(b=>b.id===current);
      c.pos=k>=0?k:Math.min(c.pos,Math.max(0,c.blocks.length-1));
      if(current===id)cicloResetTimer(false);
      save();renderCiclo();
    }
    function cicloAdd(){
      const c=cicloState(),s=state.subjects[0];if(!s)return toast('Cadastre uma disciplina antes de montar o ciclo.');
      c.blocks.push({id:uid(),subjectId:String(s.id),minutes:60});save();renderCiclo();
    }
    function cicloRebuild(btn){
      if(!cicloResetArmed){cicloResetArmed=true;btn.textContent='Confirmar: recriar ciclo';setTimeout(()=>{cicloResetArmed=false;if(btn.isConnected)btn.textContent='Recriar a partir das disciplinas';},4000);return;}
      cicloResetArmed=false;const c=cicloState();c.blocks=cicloBuildDefault();c.pos=0;cicloResetTimer(false);save();renderCiclo();toast('Ciclo recriado. O histórico de minutos foi mantido.');
    }
