  function v9Norm(s){return normalize(s||'');}
  function v9SubjectExam(s){
    try{return subjectAssessment(s)?.exam||'';}catch(e){return '';}
  }
  function v9ExamDays(s){
    const exam=v9SubjectExam(s); if(!exam)return null;
    return diffDays(exam);
  }
  function v9QuestionStats(subjectName,topicName){
    const rows=(state.questions||[]).filter(q=>v9Norm(q.subject)===v9Norm(subjectName) && (!topicName||v9Norm(q.topic)===v9Norm(topicName)));
    const total=rows.reduce((a,q)=>a+Number(q.total||0),0), correct=rows.reduce((a,q)=>a+Number(q.correct||0),0);
    return {total,correct,rate:total?Math.round(correct/total*100):null};
  }
  function v9OpenErrors(subjectName,topicName){
    return (state.recovery?.errors||[]).filter(e=>{
      const open=!/corrigido/i.test(String(e.status||''));
      const sameS=!subjectName||v9Norm(e.subject)===v9Norm(subjectName);
      const sameT=!topicName||v9Norm(e.topic)===v9Norm(topicName);
      return open&&sameS&&sameT;
    });
  }
  function v9Overdue(subjectName,topicName){
    return (state.reviews||[]).flatMap(r=>(r.items||[]).map(i=>({r,i}))).filter(x=>{
      return !x.i.done && x.i.date<todayISO() &&
        (!subjectName||v9Norm(x.r.subject)===v9Norm(subjectName)) &&
        (!topicName||v9Norm(x.r.topic)===v9Norm(topicName));
    });
  }
  function v9Priority(s){
    const days=v9ExamDays(s), topics=s.topics||[], pending=topics.filter(t=>!t.done).length;
    const q=v9QuestionStats(s.name), errs=v9OpenErrors(s.name).length, late=v9Overdue(s.name).length;
    let examScore=days===null?5:days<0?0:days<=3?35:days<=7?31:days<=14?25:days<=30?17:9;
    let weakScore=q.rate===null?6:Math.max(0,Math.round((80-q.rate)*.45));
    let pendingScore=topics.length?Math.round(pending/topics.length*15):4;
    let errorScore=Math.min(16,errs*5);
    let lateScore=Math.min(14,late*4);
    let weightScore=Math.min(10,Math.max(1,Number(s.weight||1)));
    let score=Math.min(100,examScore+weakScore+pendingScore+errorScore+lateScore+weightScore);
    const reasons=[];
    if(days!==null&&days>=0) reasons.push(days===0?'prova hoje':days===1?'prova amanhã':`prova em ${days} dias`);
    if(q.rate!==null&&q.rate<75) reasons.push(`${q.rate}% de acerto`);
    if(errs) reasons.push(`${errs} erro(s) aberto(s)`);
    if(late) reasons.push(`${late} revisão(ões) atrasada(s)`);
    if(pending) reasons.push(`${pending} tópico(s) pendente(s)`);
    return {subject:s,score,days,q,errs,late,pending,reasons};
  }
  function v9Rank(){return (state.subjects||[]).map(v9Priority).sort((a,b)=>b.score-a.score);}
  function v9NextTopic(s){
    const topics=s?.topics||[];
    const ranked=topics.map(t=>{
      const q=v9QuestionStats(s.name,t.name), errs=v9OpenErrors(s.name,t.name).length, late=v9Overdue(s.name,t.name).length;
      let score=(t.done?0:20)+(q.rate!==null?Math.max(0,75-q.rate):5)+errs*15+late*10+(t.heat==='Muito cobrado'?12:t.heat==='Cobrado'?7:0);
      return {t,q,errs,late,score};
    }).sort((a,b)=>b.score-a.score);
    return ranked[0]||null;
  }
  function v9Status(s,t){
    const q=v9QuestionStats(s.name,t.name), errs=v9OpenErrors(s.name,t.name).length, late=v9Overdue(s.name,t.name).length;
    if(errs||q.rate!==null&&q.rate<60) return {key:'critical',label:'Recuperar',icon:'!'};
    if(late||q.rate!==null&&q.rate<75) return {key:'review',label:'Revisar',icon:'↻'};
    if(t.done) return {key:'mastered',label:'Dominado',icon:'✓'};
    return {key:'pending',label:'Não iniciado',icon:'•'};
  }
  function v9FmtDay(iso){
    const d=parseISO(iso);
    return d.toLocaleDateString('pt-BR',{weekday:'short',day:'2-digit',month:'2-digit'});
  }
  function v9Agenda(days=7){
    const out=[];
    const dayNames=['DOM','SEG','TER','QUA','QUI','SEX','SAB'];
    for(let k=0;k<days;k++){
      const date=addDays(todayISO(),k), dow=dayNames[parseISO(date).getDay()];
      const items=[];
      (state.schedule||[]).filter(x=>x.day===dow).forEach(x=>items.push({kind:'Aula',time:x.time||'',title:x.activity||'Atividade',meta:[x.type,x.place].filter(Boolean).join(' · '),source:'schedule'}));
      (state.tasks||[]).filter(x=>x.date===date&&!x.done).forEach(x=>items.push({kind:x.kind||'Tarefa',time:'',title:x.title,meta:x.subject||'Geral',source:'task'}));
      (state.reviews||[]).forEach(r=>(r.items||[]).filter(i=>i.date===date&&!i.done).forEach(i=>items.push({kind:'Revisão',time:'',title:r.topic,meta:r.subject+' · '+(i.label||''),source:'review'})));
      (state.subjects||[]).forEach(s=>{const exam=v9SubjectExam(s);if(exam===date)items.push({kind:'PROVA',time:'',title:s.name,meta:'Avaliação',source:'exam'});});
      (state.recovery?.plan||[]).filter(x=>x.date===date&&x.status!=='Concluída').forEach(x=>items.push({kind:x.type||'Plano',time:'',title:x.title||x.topic||'Bloco de estudo',meta:x.subject||'',source:'plan'}));
      items.sort((a,b)=>(a.time||'99:99').localeCompare(b.time||'99:99'));
      out.push({date,dow,items});
    }
    return out;
  }
