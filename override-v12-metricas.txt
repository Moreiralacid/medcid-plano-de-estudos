/* ===== MEDCID v12 — Painel de desempenho na aba Evolução =====
   Tela própria do MEDCID. Lê somente dados que o Planner já registra:
   plano de 7 dias, revisões espaçadas e sessões do Banco de questões.
   Não altera, migra nem apaga dados. Para desativar, remova este arquivo do index.html. */
(function(){
  const V12='2026-11-16.1';
  const V12_WEAK=70;          // mesmo corte de "ponto fraco" usado no Banco de questões
  const V12_SMALL_SAMPLE=20;  // abaixo disso, o percentual ainda oscila muito
  let v12Period='7';
  const v12OpenSubjects=new Set();

  /* ---------- datas ---------- */
  const v12InRange=(d,w)=>!!d&&d>=w.start&&d<=w.end;
  function v12PlanDate(t){return t.date||addDays(state.recovery?.profile?.start||todayISO(),Number(t.day||0));}
  function v12Earliest(){
    const t=todayISO();let min=null;
    const see=d=>{if(d&&d<=t&&(!min||d<min))min=d;};
    (state.questions||[]).forEach(q=>see(q.date));
    (state.recovery?.plan||[]).forEach(p=>see(v12PlanDate(p)));
    (state.reviews||[]).forEach(r=>(r.items||[]).forEach(i=>see(i.date)));
    return min;
  }
  function v12Window(period){
    const end=todayISO();
    if(period==='all'){
      const start=v12Earliest()||end;
      return {start,end,len:Math.max(1,diffDays(end,start)+1),prev:null};
    }
    const len=Number(period),start=addDays(end,-(len-1)),prevEnd=addDays(start,-1);
    return {start,end,len,prev:{start:addDays(prevEnd,-(len-1)),end:prevEnd,len}};
  }
  function v12WeekBounds(offset){
    const d=parseISO(todayISO()),monday=new Date(d);
    monday.setDate(d.getDate()-((d.getDay()+6)%7)+offset*7);
    const start=monday.toISOString().slice(0,10);
    return {start,end:addDays(start,6)};
  }

  /* ---------- métricas ---------- */
  function v12Plan(w){
    const tasks=(state.recovery?.plan||[]).filter(t=>t.source!=='class'&&v12InRange(v12PlanDate(t),w));
    const done=tasks.filter(t=>t.status==='Concluída').length,partial=tasks.filter(t=>t.status==='Parcial').length;
    return {total:tasks.length,done,partial,pct:tasks.length?Math.round((done+partial*.5)/tasks.length*100):null};
  }
  function v12Reviews(w){
    const t=todayISO();let due=0,onTime=0,late=0,pending=0;
    (state.reviews||[]).forEach(r=>(r.items||[]).forEach(i=>{
      if(!v12InRange(i.date,w))return;
      if(i.date===t&&!i.done)return; // ainda dá tempo hoje: não conta como atraso
      due++;
      if(i.done){if(!i.completedAt||i.completedAt<=i.date)onTime++;else late++;}
      else pending++;
    }));
    return {due,onTime,late,pending,pct:due?Math.round(onTime/due*100):null};
  }
  function v12Questions(w){
    const qs=(state.questions||[]).filter(q=>v12InRange(q.date,w));
    const total=qs.reduce((a,q)=>a+Number(q.total||0),0),correct=qs.reduce((a,q)=>a+Number(q.correct||0),0);
    return {qs,total,correct,acc:total?Math.round(correct/total*100):null,sessions:qs.length,days:new Set(qs.map(q=>q.date)).size};
  }
  function v12Goal(){return Math.max(1,Number(state.config?.dailyQuestionsGoal||30));}
  function v12PerDay(){
    const map={};(state.questions||[]).forEach(q=>{if(q.date)map[q.date]=(map[q.date]||0)+Number(q.total||0);});return map;
  }
  function v12Streaks(){
    const per=v12PerDay(),goal=v12Goal(),t=todayISO();
    let cur=0,d=per[t]>=goal?t:addDays(t,-1);
    while((per[d]||0)>=goal){cur++;d=addDays(d,-1);}
    const days=Object.keys(per).sort();let best=0,run=0,prev=null;
    days.forEach(day=>{
      if((per[day]||0)>=goal){run=(prev&&addDays(prev,1)===day)?run+1:1;prev=day;best=Math.max(best,run);}
      else{run=0;prev=null;}
    });
    return {cur,best:Math.max(best,cur)};
  }
  function v12BySubject(w){
    const map={};
    v12Questions(w).qs.forEach(q=>{
      const sk=normalize(q.subject).trim()||'(sem disciplina)';
      if(!map[sk])map[sk]={key:sk,name:q.subject||'(sem disciplina)',total:0,correct:0,topics:{}};
      const s=map[sk];s.total+=Number(q.total||0);s.correct+=Number(q.correct||0);
      const tk=normalize(q.topic).trim()||'(sem assunto)';
      if(!s.topics[tk])s.topics[tk]={name:q.topic||'(sem assunto)',total:0,correct:0};
      s.topics[tk].total+=Number(q.total||0);s.topics[tk].correct+=Number(q.correct||0);
    });
    const rate=x=>x.total?Math.round(x.correct/x.total*100):0;
    return Object.values(map).map(s=>({...s,rate:rate(s),topics:Object.values(s.topics).map(t=>({...t,rate:rate(t)})).sort((a,b)=>a.rate-b.rate)}))
      .sort((a,b)=>a.rate-b.rate||b.total-a.total);
  }

  /* ---------- apresentação ---------- */
  function v12Tone(r){return r===null||r===undefined?'none':r<50?'bad':r<V12_WEAK?'warn':'good';}
  function v12Delta(curr,prev){
    if(curr===null||prev===null||prev===undefined)return '';
    const d=curr-prev;if(d===0)return '<small class="v12-delta">= período anterior</small>';
    return `<small class="v12-delta ${d>0?'up':'down'}">${d>0?'▲ +':'▼ '}${d} pp vs. período anterior</small>`;
  }
  function v12Kpi(label,value,tone,sub,extra=''){
    return `<div class="v12-kpi tone-${tone}"><span class="v12-kpi-label">${label}</span><strong>${value}</strong><small>${sub}</small>${extra}</div>`;
  }
  function v12PeriodLabel(w){return v12Period==='all'?`desde ${fmtDate(w.start)}`:`últimos ${w.len} dias`;}

  function v12RenderKpis(w){
    const plan=v12Plan(w),rev=v12Reviews(w),q=v12Questions(w),goal=v12Goal();
    const prevQ=w.prev?v12Questions(w.prev):null;
    const goalPct=Math.round(q.total/(goal*w.len)*100),avg=(q.total/w.len).toFixed(q.total/w.len<10?1:0);
    const backlog=overdueReviews().length;
    return [
      v12Kpi('Execução do plano',plan.pct===null?'—':plan.pct+'%',v12Tone(plan.pct),
        plan.total?`${plan.done} concluídos · ${plan.partial} parciais de ${plan.total} blocos`:'Nenhum bloco planejado no período'),
      v12Kpi('Revisões em dia',rev.pct===null?'—':rev.pct+'%',v12Tone(rev.pct),
        rev.due?`${rev.onTime} no prazo · ${rev.late} com atraso · ${rev.pending} pendentes`:'Nenhuma revisão vencida no período',
        backlog?`<small class="v12-alert">${backlog} ${backlog>1?'revisões atrasadas':'revisão atrasada'} na fila</small>`:''),
      v12Kpi('Questões × meta',q.total?goalPct+'%':'—',v12Tone(q.total?goalPct:null),
        `${q.total} feitas · média ${avg}/dia · meta ${goal}/dia`),
      v12Kpi('Acerto em questões',q.acc===null?'—':q.acc+'%',v12Tone(q.acc),
        q.total?`${q.correct}/${q.total} acertos${q.total<V12_SMALL_SAMPLE?' · amostra pequena':''}`:'Sem sessões registradas',
        prevQ?v12Delta(q.acc,prevQ.acc):'')
    ].join('');
  }

  function v12RenderDaily(){
    const per=v12PerDay(),goal=v12Goal(),days=Array.from({length:14},(_,i)=>addDays(todayISO(),i-13));
    const max=Math.max(goal*1.25,...days.map(d=>per[d]||0)),goalPos=Math.round(goal/max*100);
    const met=days.filter(d=>(per[d]||0)>=goal).length,st=v12Streaks();
    const wd=['D','S','T','Q','Q','S','S'];
    const cols=days.map(d=>{
      const n=per[d]||0,h=n?Math.max(4,Math.round(n/max*100)):0,cls=n>=goal?'met':n?'part':'zero';
      return `<div class="v12-day ${cls}${d===todayISO()?' today':''}" title="${fmtDate(d)}: ${n} questões">
        <span class="v12-day-val">${n||''}</span><div class="v12-day-track"><i style="height:${h}%"></i></div>
        <small>${wd[parseISO(d).getDay()]}<br>${d.slice(8)}</small></div>`;
    }).join('');
    return `<article class="card v12-card">
      <div class="v12-card-head"><div><h3>Questões × meta diária</h3><p>Últimos 14 dias · meta de ${goal} por dia</p></div>
        <div class="v12-streak"><strong>${st.cur}</strong><span>dia${st.cur===1?'':'s'} seguidos na meta</span><small>melhor: ${st.best}</small></div></div>
      <div class="v12-daily" role="img" aria-label="Questões por dia nos últimos 14 dias. Meta batida em ${met} de 14 dias.">
        <div class="v12-goal-line" style="bottom:calc(34px + (100% - 56px) * ${(goalPos/100).toFixed(3)})"><span>meta</span></div>${cols}</div>
      <p class="v12-foot">Meta batida em <b>${met}</b> de 14 dias. Altere a meta diária em Configurações.</p>
    </article>`;
  }

  function v12RenderWeekly(){
    const weeks=Array.from({length:8},(_,i)=>{
      const b=v12WeekBounds(i-7),q=v12Questions(b);return {b,acc:q.acc,total:q.total};
    });
    const W=320,H=150,L=30,R=10,T=14,B=26,iw=W-L-R,ih=H-T-B;
    const x=i=>L+iw*(i/7),y=v=>T+ih*(1-v/100);
    let path='',open=false;
    weeks.forEach((wk,i)=>{if(wk.acc===null){open=false;return;}path+=(open?'L':'M')+x(i).toFixed(1)+' '+y(wk.acc).toFixed(1)+' ';open=true;});
    const grid=[0,50,100].map(v=>`<line x1="${L}" x2="${W-R}" y1="${y(v)}" y2="${y(v)}" class="v12-grid"/><text x="${L-6}" y="${y(v)+4}" text-anchor="end" class="v12-axis">${v}</text>`).join('');
    const dots=weeks.map((wk,i)=>wk.acc===null?
      `<text x="${x(i)}" y="${y(4)}" text-anchor="middle" class="v12-axis">—</text>`:
      `<circle cx="${x(i)}" cy="${y(wk.acc)}" r="4.5" class="v12-dot tone-${v12Tone(wk.acc)}"><title>Semana de ${fmtDate(wk.b.start)}: ${wk.acc}% (${wk.total} questões)</title></circle>`).join('');
    const labels=weeks.map((wk,i)=>`<text x="${x(i)}" y="${H-8}" text-anchor="middle" class="v12-axis">${wk.b.start.slice(8)}/${wk.b.start.slice(5,7)}</text>`).join('');
    const withData=weeks.filter(w=>w.acc!==null),last=withData[withData.length-1],first=withData[0];
    let summary='Registre sessões no Banco de questões para ver a curva.';
    if(withData.length===1)summary=`Só uma semana com dados (${last.acc}%). A tendência aparece a partir da segunda.`;
    else if(withData.length>1){const d=last.acc-first.acc;summary=`${d>0?'Subiu':d<0?'Caiu':'Estável'} ${d?Math.abs(d)+' pp ':''}entre a primeira e a última semana com dados.`;}
    const desc=weeks.map(w=>`${fmtDate(w.b.start)}: ${w.acc===null?'sem dados':w.acc+'%'}`).join('; ');
    return `<article class="card v12-card">
      <div class="v12-card-head"><div><h3>Evolução semanal de acertos</h3><p>Últimas 8 semanas · linha tracejada = ${V12_WEAK}%</p></div></div>
      <svg class="v12-weekly" viewBox="0 0 ${W} ${H}" role="img" aria-label="Acerto semanal em questões. ${desc}">
        ${grid}<line x1="${L}" x2="${W-R}" y1="${y(V12_WEAK)}" y2="${y(V12_WEAK)}" class="v12-target"/>
        <path d="${path}" class="v12-line"/>${dots}${labels}</svg>
      <p class="v12-foot">${summary}</p>
    </article>`;
  }

  function v12RenderSubjects(w){
    const rows=v12BySubject(w);
    if(!rows.length)return `<article class="card v12-card"><div class="v12-card-head"><div><h3>Acertos por disciplina e assunto</h3><p>${v12PeriodLabel(w)}</p></div></div>
      <div class="empty">Nenhuma sessão de questões neste período.</div>
      <button type="button" class="btn soft" data-v12-go="questoes">Registrar sessão de questões</button></article>`;
    const bar=(name,x,extra='')=>`<div class="v12-bar-head"><span>${name}</span><b class="tone-${v12Tone(x.rate)}">${x.rate}%</b></div>
      <div class="v12-bar"><i class="tone-${v12Tone(x.rate)}" style="width:${x.rate}%"></i></div>
      <small class="v12-bar-sub">${x.correct}/${x.total} acertos${x.total<V12_SMALL_SAMPLE?' · <em>amostra pequena</em>':''}${extra}</small>`;
    const weak=rows.filter(r=>r.rate<V12_WEAK&&r.total>=V12_SMALL_SAMPLE).length;
    const list=rows.map(s=>{
      const open=v12OpenSubjects.has(s.key);
      return `<div class="v12-subject${open?' open':''}">
        <button type="button" class="v12-subject-btn" data-v12-subject="${escapeHtml(s.key)}" aria-expanded="${open}">
          ${bar(escapeHtml(s.name),s,` · ${s.topics.length} assunto${s.topics.length>1?'s':''}`)}<span class="v12-caret" aria-hidden="true">▸</span></button>
        ${open?`<div class="v12-topics">${s.topics.map(t=>`<div class="v12-topic">${bar(escapeHtml(t.name),t)}</div>`).join('')}</div>`:''}
      </div>`;
    }).join('');
    return `<article class="card v12-card v12-wide">
      <div class="v12-card-head"><div><h3>Acertos por disciplina e assunto</h3><p>${v12PeriodLabel(w)} · mais fracos primeiro · toque para abrir os assuntos</p></div>
        <span class="badge ${weak?'orange':'green'}">${weak} abaixo de ${V12_WEAK}%</span></div>
      <div class="v12-subjects">${list}</div>
    </article>`;
  }

  function v12Render(){
    const box=$('#v12Metrics');if(!box)return;
    const w=v12Window(v12Period),q=v12Questions(w);
    $$('#v12Metrics [data-v12-period]').forEach(b=>b.setAttribute('aria-pressed',String(b.dataset.v12Period===v12Period)));
    $('#v12Kpis').innerHTML=v12RenderKpis(w);
    $('#v12Charts').innerHTML=v12RenderDaily()+v12RenderWeekly();
    $('#v12Subjects').innerHTML=v12RenderSubjects(w);
    $('#v12DataNote').innerHTML=`Base do período (${v12PeriodLabel(w)}): <b>${q.sessions}</b> ${q.sessions===1?'sessão':'sessões'} de questões em <b>${q.days}</b> dia${q.days===1?'':'s'}. `+
      `Questões e acertos dependem do que você registra no Banco de questões; execução e revisões vêm do plano e da agenda.`;
  }

  function v12Mount(){

    const head=prog.querySelector('.section-head h2');if(head)head.textContent='Evolução & métricas';
    const sub=prog.querySelector('.section-head p:last-child');if(sub)sub.textContent='Plano, revisões e questões num só painel — os números saem do que você registra no MEDCID.';
    const oldToolbar=prog.querySelector(':scope > .plan-toolbar');if(oldToolbar)oldToolbar.hidden=true; // números agora vivem no painel
    const box=document.createElement('section');box.id='v12Metrics';box.setAttribute('aria-label','Painel de desempenho');
    box.innerHTML=`
      <div class="v12-top"><div><h2>Painel de desempenho</h2><p>Execução, revisões e questões no mesmo lugar.</p></div>
        <div class="v12-periods" role="group" aria-label="Período">
          <button type="button" data-v12-period="7" aria-pressed="true">7 dias</button>
          <button type="button" data-v12-period="30" aria-pressed="false">30 dias</button>
          <button type="button" data-v12-period="all" aria-pressed="false">Tudo</button>
        </div></div>
      <div id="v12Kpis" class="v12-kpis"></div>
