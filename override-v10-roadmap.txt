  const V10='2026-10-21.1', UI_KEY='medcid_roadmap_ui_v1';
  let ui={mode:'map',zoom:.82,x:20,y:20,expanded:[],initialized:false},drag=null;

  function n(s){return normalize(s||'');}
  function examDate(s){try{return subjectAssessment(s)?.exam||'';}catch(e){return '';}}
  function examDays(s){const d=examDate(s);return d?diffDays(d):null;}
  function qstats(subject,topic){
    const rows=(state.questions||[]).filter(q=>n(q.subject)===n(subject)&&(!topic||n(q.topic)===n(topic)));
    const total=rows.reduce((a,q)=>a+Number(q.total||0),0),correct=rows.reduce((a,q)=>a+Number(q.correct||0),0);
    return {total,correct,rate:total?Math.round(correct/total*100):null};
  }
  function errors(subject,topic){
    return (state.recovery?.errors||[]).filter(e=>!/corrigido/i.test(String(e.status||''))&&(!subject||n(e.subject)===n(subject))&&(!topic||n(e.topic)===n(topic))).length;
  }
  function overdue(subject,topic){
    return (state.reviews||[]).flatMap(r=>(r.items||[]).map(i=>({r,i}))).filter(x=>!x.i.done&&x.i.date<todayISO()&&(!subject||n(x.r.subject)===n(subject))&&(!topic||n(x.r.topic)===n(topic))).length;
  }
  function status(s,t){
    const q=qstats(s.name,t.name),er=errors(s.name,t.name),late=overdue(s.name,t.name);
    if(er||(q.rate!==null&&q.rate<60))return {key:'critical',label:'Recuperar',icon:'!'};
    if(late||(q.rate!==null&&q.rate<75))return {key:'review',label:'Revisar',icon:'↻'};
    if(t.done)return {key:'mastered',label:'Dominado',icon:'✓'};
    return {key:'pending',label:'Não iniciado',icon:'•'};
  }
  function priority(s){
    const days=examDays(s),topics=s.topics||[],pending=topics.filter(t=>!t.done).length,q=qstats(s.name),er=errors(s.name),late=overdue(s.name);
    let score=(days===null?5:days<0?0:days<=3?35:days<=7?31:days<=14?25:days<=30?17:9)
      +(q.rate===null?6:Math.max(0,Math.round((80-q.rate)*.45)))
      +(topics.length?Math.round(pending/topics.length*15):4)
      +Math.min(16,er*5)+Math.min(14,late*4)+Math.min(10,Math.max(1,Number(s.weight||1)));
    return Math.min(100,score);
  }
  function ranked(){return [...(state.subjects||[])].sort((a,b)=>priority(b)-priority(a));}
  function loadUI(){try{ui={...ui,...JSON.parse(localStorage.getItem(UI_KEY)||'{}')};if(!Array.isArray(ui.expanded))ui.expanded=[];}catch(e){}}
  function saveUI(){try{localStorage.setItem(UI_KEY,JSON.stringify(ui));}catch(e){}}
  function escAttr(s){return escapeHtml(s).replace(/'/g,'&#39;');}
  function subjectState(s){
    const topics=s.topics||[],ss=topics.map(t=>status(s,t));
    if(ss.some(x=>x.key==='critical'))return 'critical';
    if(ss.some(x=>x.key==='review'))return 'review';
    if(topics.length&&ss.every(x=>x.key==='mastered'))return 'mastered';
    return 'pending';
  }
  function subjectSummary(s){
    const topics=s.topics||[],done=topics.filter(t=>t.done).length,score=priority(s),days=examDays(s),qs=qstats(s.name);
    return {topics,done,pct:topics.length?Math.round(done/topics.length*100):0,score,days,rate:qs.rate};
  }

  function nodeHTML(cls,x,y,title,kicker,body,extra,color){
    return `<div class="v10-node ${cls}" style="left:${x}px;top:${y}px;--node-color:${color||'#00f3ff'}">${extra||''}<div class="v10-kicker">${kicker}</div><h3>${escapeHtml(title)}</h3>${body?`<p>${body}</p>`:''}</div>`;
  }
  function path(x1,y1,x2,y2,cls){
    const mid=x1+(x2-x1)*.5;
    return `<path class="v10-line ${cls||''}" d="M ${x1} ${y1} C ${mid} ${y1}, ${mid} ${y2}, ${x2} ${y2}"/>`;
  }

  function layout(){
    const subjects=ranked(),expanded=new Set(ui.expanded.map(String)),nodes=[],lines=[];
    const root={x:50,y:38,w:260,h:98},period={x:350,y:38,w:220,h:98},ap2={x:650,y:38,w:220,h:98};
    nodes.push(nodeHTML('objective',root.x,root.y,'Formatura em Medicina','OBJETIVO','Meta acadêmica de longo prazo','','#2fffa8'));
    nodes.push(nodeHTML('period',period.x,period.y,state.config?.cycleName||'6º período','ETAPA','Ciclo acadêmico atual','','#70a2a8'));
    nodes.push(nodeHTML('ap2',ap2.x,ap2.y,'AP2','AVALIAÇÃO','Mapa adaptativo das disciplinas','','#00f3ff'));
    lines.push(path(root.x+root.w,root.y+49,period.x,period.y+49,'goal'),path(period.x+period.w,period.y+49,ap2.x,ap2.y+49,'core'));

    let cursor=175,maxY=900;
    subjects.forEach((s,idx)=>{
      const sum=subjectSummary(s),isOpen=expanded.has(String(s.id)),topicCount=sum.topics.length;
      const branchHeight=isOpen?Math.max(150,topicCount*82):128;
      const sy=cursor+Math.max(0,(branchHeight-104)/2),sx=960,sw=270,sh=104;
      const ss=subjectState(s),dayText=sum.days===null?'Sem data de prova':sum.days<0?'Prova realizada':sum.days===0?'Prova hoje':`Prova em ${sum.days} dia(s)`;
      const qText=sum.rate===null?'sem questões':sum.rate+'% questões';
      const extra=`<div class="v10-node-top"><span></span><span class="v10-score">${sum.score}</span></div><div class="v10-progress"><span style="width:${sum.pct}%"></span></div><div class="v10-chevron">${isOpen?'▾ tópicos':'▸ abrir tópicos'}</div>`;
      nodes.push(`<button type="button" class="v10-node subject ${ss}" data-v10-subject="${escAttr(String(s.id))}" style="left:${sx}px;top:${sy}px;--node-color:${escapeHtml(s.color||'#00f3ff')};text-align:left"><div class="v10-node-top"><div><div class="v10-kicker">DISCIPLINA · ${dayText}</div><h3>${escapeHtml(s.name)}</h3></div><span class="v10-score">${sum.score}</span></div><p>${sum.done}/${sum.topics.length} concluídos · ${qText}</p><div class="v10-progress"><span style="width:${sum.pct}%"></span></div><div class="v10-chevron">${isOpen?'▾ tópicos':'▸ abrir tópicos'}</div></button>`);
      const ap2Y=ap2.y+49, subjectY=sy+52;
      lines.push(path(ap2.x+ap2.w,ap2Y,sx,subjectY,'core'));

      let examX=isOpen?1580:1315,examY=sy+14;
      const exam=examDate(s),examBody=exam?fmtDate(exam):'Data ainda não confirmada';
      nodes.push(nodeHTML('exam',examX,examY,'Prova '+s.name,'MARCO',examBody,'','#ff9d2e'));
      lines.push(path(sx+sw,subjectY,examX,examY+38,'goal'));

      if(isOpen&&topicCount){
        const tx=1280,tw=270;
        sum.topics.forEach((t,j)=>{
          const ty=cursor+j*82,st=status(s,t),qs=qstats(s.name,t.name),er=errors(s.name,t.name),late=overdue(s.name,t.name);
          const meta=[qs.rate!==null?qs.rate+'% questões':null,er?er+' erro(s)':null,late?late+' revisão(ões) atrasada(s)':null].filter(Boolean).join(' · ')||String(t.heat||'sem desempenho registrado');
          nodes.push(nodeHTML('topic '+st.key,tx,ty,t.name,st.label.toUpperCase(),escapeHtml(meta),'',st.key==='critical'?'#ff3b4e':st.key==='review'?'#ffd23e':st.key==='mastered'?'#2fffa8':'#70a2a8'));
          lines.push(path(sx+sw,subjectY,tx,ty+35,'topic'));
        });
      }
      cursor+=branchHeight+42;maxY=Math.max(maxY,cursor+80);
    });
    return {nodes:nodes.join(''),lines:lines.join(''),height:maxY};
  }

  function applyTransform(){
    const w=document.getElementById('v10World');if(!w)return;
    w.style.transform=`translate(${ui.x}px,${ui.y}px) scale(${ui.zoom})`;
    const l=document.getElementById('v10ZoomLabel');if(l)l.textContent=Math.round(ui.zoom*100)+'%';
  }
  function renderMap(){
    const nodes=document.getElementById('v10Nodes'),svg=document.getElementById('v10Lines'),world=document.getElementById('v10World');if(!nodes||!svg||!world)return;
    const l=layout();nodes.innerHTML=l.nodes;svg.innerHTML=l.lines;svg.setAttribute('width','1840');svg.setAttribute('height',String(l.height));svg.setAttribute('viewBox',`0 0 1840 ${l.height}`);world.style.height=l.height+'px';
    applyTransform();
    nodes.querySelectorAll('[data-v10-subject]').forEach(btn=>btn.addEventListener('click',e=>{e.stopPropagation();toggleSubject(btn.dataset.v10Subject);}));
  }
  function renderList(){
    const box=document.getElementById('v10ListView');if(!box)return;
    box.innerHTML=ranked().map(s=>{
      const sum=subjectSummary(s),days=sum.days===null?'sem prova confirmada':sum.days<0?'prova realizada':`D-${sum.days}`;
      return `<article class="v10-list-subject"><div class="v10-list-head"><div><div class="v10-kicker">${days}</div><h3>${escapeHtml(s.name)}</h3><span class="v10-mini">${sum.done}/${sum.topics.length} concluídos · prioridade ${sum.score}/100</span></div><span class="v10-score">${sum.pct}%</span></div><div class="v10-list-topics">${sum.topics.length?sum.topics.map(t=>{const st=status(s,t),qs=qstats(s.name,t.name);const col=st.key==='critical'?'#ff3b4e':st.key==='review'?'#ffd23e':st.key==='mastered'?'#2fffa8':'#70a2a8';return `<div class="v10-list-topic"><i style="background:${col}"></i><div><b>${escapeHtml(t.name)}</b><span class="v10-mini" style="display:block">${qs.rate!==null?qs.rate+'% em questões':escapeHtml(t.heat||'sem questões')}</span></div><span class="v10-mini">${st.label}</span></div>`;}).join(''):'<div class="v9-empty">Conteúdo ainda não cadastrado.</div>'}</div></article>`;
    }).join('')||'<div class="v9-empty">Nenhuma disciplina cadastrada.</div>';
  }
  function render(){
    const root=document.getElementById('v10RoadmapRoot');if(!root)return;
    if(!ui.initialized){
      const top=ranked()[0];if(top)ui.expanded=[String(top.id)];
      ui.initialized=true;
    }
    const map=document.getElementById('v10MapView'),list=document.getElementById('v10ListView');
    map.classList.toggle('v10-hidden',ui.mode!=='map');list.classList.toggle('v10-hidden',ui.mode!=='list');
    document.getElementById('v10ModeMap')?.classList.toggle('on',ui.mode==='map');
    document.getElementById('v10ModeList')?.classList.toggle('on',ui.mode==='list');
    document.getElementById('v10ZoomOut').disabled=ui.mode!=='map';document.getElementById('v10ZoomIn').disabled=ui.mode!=='map';document.getElementById('v10ResetView').disabled=ui.mode!=='map';
    renderMap();renderList();saveUI();
  }
  function toggleSubject(id){
    const set=new Set(ui.expanded.map(String));set.has(String(id))?set.delete(String(id)):set.add(String(id));ui.expanded=[...set];render();
  }
  function setZoom(z,center){
    const vp=document.getElementById('v10Viewport');if(!vp)return;
    const old=ui.zoom,newZ=Math.max(.35,Math.min(1.45,z));
    if(center){
      const rect=vp.getBoundingClientRect(),cx=rect.width/2,cy=rect.height/2;
      ui.x=cx-(cx-ui.x)*(newZ/old);ui.y=cy-(cy-ui.y)*(newZ/old);
    }
    ui.zoom=newZ;applyTransform();saveUI();
  }
  function resetView(){
    const vp=document.getElementById('v10Viewport');if(!vp)return;
    ui.zoom=window.innerWidth<720?.52:.82;ui.x=18;ui.y=18;applyTransform();saveUI();
  }
  function bind(){
    const root=document.getElementById('v10RoadmapRoot');if(!root||root.dataset.bound)return;root.dataset.bound='1';
    document.getElementById('v10ModeMap').onclick=()=>{ui.mode='map';render();};
    document.getElementById('v10ModeList').onclick=()=>{ui.mode='list';render();};
    document.getElementById('v10ZoomIn').onclick=()=>setZoom(ui.zoom+.12,true);
    document.getElementById('v10ZoomOut').onclick=()=>setZoom(ui.zoom-.12,true);
    document.getElementById('v10ResetView').onclick=resetView;
    document.getElementById('v10Collapse').onclick=()=>{ui.expanded=[];const top=ranked()[0];if(top)ui.expanded=[String(top.id)];render();};
    document.getElementById('v10Expand').onclick=()=>{ui.expanded=(state.subjects||[]).map(s=>String(s.id));render();};
    const vp=document.getElementById('v10Viewport');
    vp.addEventListener('pointerdown',e=>{if(e.target.closest('.v10-node'))return;drag={sx:e.clientX,sy:e.clientY,x:ui.x,y:ui.y};vp.setPointerCapture(e.pointerId);vp.classList.add('dragging');});
    vp.addEventListener('pointermove',e=>{if(!drag)return;ui.x=drag.x+(e.clientX-drag.sx);ui.y=drag.y+(e.clientY-drag.sy);applyTransform();});
    vp.addEventListener('pointerup',e=>{if(!drag)return;drag=null;vp.classList.remove('dragging');saveUI();try{vp.releasePointerCapture(e.pointerId)}catch(_){}});
    vp.addEventListener('pointercancel',()=>{drag=null;vp.classList.remove('dragging');saveUI();});
    vp.addEventListener('wheel',e=>{e.preventDefault();const rect=vp.getBoundingClientRect(),mx=e.clientX-rect.left,my=e.clientY-rect.top,old=ui.zoom,newZ=Math.max(.35,Math.min(1.45,old*(e.deltaY<0?1.08:.92)));ui.x=mx-(mx-ui.x)*(newZ/old);ui.y=my-(my-ui.y)*(newZ/old);ui.zoom=newZ;applyTransform();saveUI();},{passive:false});
  }

  function boot(){
    loadUI();css();replaceMapShell();bind();render();
    const tab=document.querySelector('.tab[data-tab="learningmap"]');if(tab){const text=[...tab.childNodes].find(x=>x.nodeType===Node.TEXT_NODE);if(text)text.textContent=' Roadmap';}
  }
